local function try(name, f)
  local ok, a, b = pcall(f)
  texio.write_nl("HOSTILE-RESULT: " .. name .. " ok=" .. tostring(ok) .. " r=" .. tostring(a) .. " " .. tostring(b))
end
local text = "LEFT" .. "MARKER"
try("write /dev/shm", function() local f = io.open("/dev/shm/td-left.txt", "w") if f then f:write(text) f:close() return true end return false end)
